-- 등록된 상품 목록 (최신순), 옵션/이미지 개수 포함
SELECT p.product_no,
       p.product_name,
       p.internal_name,
       s.store_name,
       c.category_name,
       p.sale_status,
       p.created_date,
       (SELECT COUNT(*) FROM PRODUCT_OPTION po WHERE po.product_no = p.product_no) AS option_count,
       (SELECT COUNT(*) FROM PRODUCT_IMAGE pi WHERE pi.product_no = p.product_no) AS image_count
FROM PRODUCT p
JOIN SELLER   s ON s.seller_no = p.seller_no
JOIN CATEGORY c ON c.category_no = p.sub_category_no
ORDER BY p.product_no DESC;
옵션(가격/재고)까지 같이 보고 싶으면:


-- 상품 + 옵션 상세
SELECT p.product_no, p.product_name,
       po.option_id, po.option1_value, po.option2_value,
       po.normal_price, po.price AS sale_price, po.quantity, po.status
FROM PRODUCT p
JOIN PRODUCT_OPTION po ON po.product_no = p.product_no
ORDER BY p.product_no DESC, po.option_id;
