# 테이블 생성 및 상품등록 - 신창만 

- TOCK_HISTORY 변동유형·수량 => check제약 확인하기 => 확인완료
- 재고 관리 프로시저-> 판매자별 각 상품별 재고수량 확인쿼리 => 완료
- 회원,판매자 => 아이디 유니크 값 추가 => 완료
- 장바구니보기, 찜하기 => web page만들것

⚠ **2026-08-23 갱신**: 이 파일의 `PRODUCT`/`PRODUCT_OPTION`/카테고리 관련 쿼리를
`CreataTableDDL.md`의 새 스키마(컬럼 확장 + 자기참조 `CATEGORY` 테이블)에 맞춰 전부 수정함.
아래 세 가지가 핵심 변경점입니다.

1. `PRODUCT`가 7개 컬럼 → 40개 이상으로 늘어나서, `VALUES`에 값만 순서대로 나열하던 방식 대신
   컬럼명을 전부 명시하는 `INSERT INTO PRODUCT (컬럼...) VALUES (값...)` 형태로 바꿈
   (컬럼이 많고 대부분 NOT NULL이라 순서를 틀리면 바로 에러가 나서, 컬럼명을 쓰는 쪽이 훨씬 안전함)
2. `PRODUCT_OPTION`은 `PRICE` 하나였던 컬럼이 `normal_price`(정상가)/`sale_price`(판매가) 둘로 나뉨
3. `MAIN_CATEGORY`/`MID_CATEGORY`/`SUB_CATEGORY` 3단계 테이블 조인 → 자기참조 `CATEGORY` 테이블 하나를
   3번 셀프 조인(`c1`=대분류, `c2`=중분류, `c3`=소분류)하는 방식으로 바꿈

⚠ **옵션 없는 상품(예: 생수)의 재고 관리 방식이 근본적으로 바뀝니다** — 아래 "상품등록 (트리거·시퀀스·샘플데이터)" 절
맨 앞의 설명을 꼭 읽어보세요. `UP_PRODUCT_IN` 프로시저가 `OPTION_ID` 기준이라, 옵션이 아예 없는 상품(`option_yn='N'`)은
이 프로시저로 입고 처리가 안 됩니다.

### 상품목록보기(판매자기준)

```sql
-- 1번 판매자가 등록한 상품조회
-- (SUB_CATEGORY/MID_CATEGORY/MAIN_CATEGORY 3단계 조인 → 자기참조 CATEGORY 셀프조인으로 변경)
SELECT p.*,
       s.store_name,
       c1.category_name AS main_category_name,
       c2.category_name AS mid_category_name,
       c3.category_name AS sub_category_name
FROM PRODUCT p
    JOIN SELLER s
      ON p.seller_no = s.seller_no
    JOIN CATEGORY c3
      ON p.category_no = c3.category_no
    LEFT JOIN CATEGORY c2
      ON c2.category_no = c3.parent_category_no
    LEFT JOIN CATEGORY c1
      ON c1.category_no = c2.parent_category_no
WHERE p.seller_no = 1;

-- 1번 판매자의 각 상품별 재고 현황조회
-- (product_price→base_price, product_name→display_name, po.PRICE→normal_price/sale_price로 분리)
SELECT p.product_no, p.display_name, p.product_desc, p.base_price,
       po.normal_price   AS option_normal_price,
       po.sale_price     AS option_sale_price,
       po.quantity       AS option_quantity,
       po.OPTION_ID,
       po.OPTION1_TYPE,
       po.OPTION1_VALUE,
       po.OPTION2_TYPE,
       po.OPTION2_VALUE,
       po.OPTION3_TYPE,
       po.OPTION3_VALUE,
       po.STATUS         AS option_status
FROM PRODUCT p
JOIN SELLER s
  ON p.seller_no = s.seller_no
LEFT JOIN PRODUCT_OPTION po
  ON p.product_no = po.product_no
WHERE p.seller_no = 1;
```

### 카테고리보기

```sql
-- (자기참조 CATEGORY를 3번 셀프조인: c1=대분류, c2=중분류, c3=소분류)
SELECT c1.category_no   AS main_category_no,
       c1.category_name AS main_category_name,
       c2.category_no   AS mid_category_no,
       c2.category_name AS mid_category_name,
       c3.category_no   AS sub_category_no,
       c3.category_name AS sub_category_name
FROM CATEGORY c1
    LEFT JOIN CATEGORY c2 ON c2.parent_category_no = c1.category_no AND c2.category_level = 2
    LEFT JOIN CATEGORY c3 ON c3.parent_category_no = c2.category_no AND c3.category_level = 3
WHERE c1.category_level = 1
ORDER BY c1.category_no, c2.category_no, c3.category_no;

-- QA를 위한 카테고리별 상품갯수
SELECT c1.category_no   AS main_category_no,
       c1.category_name AS main_category_name,
       c2.category_no   AS mid_category_no,
       c2.category_name AS mid_category_name,
       c3.category_no   AS sub_category_no,
       c3.category_name AS sub_category_name,
       COUNT(p.product_no) AS product_count
FROM CATEGORY c1
    LEFT JOIN CATEGORY c2 ON c2.parent_category_no = c1.category_no AND c2.category_level = 2
    LEFT JOIN CATEGORY c3 ON c3.parent_category_no = c2.category_no AND c3.category_level = 3
    LEFT JOIN PRODUCT p ON c3.category_no = p.category_no
WHERE c1.category_level = 1
GROUP BY c1.category_no, c1.category_name,
         c2.category_no, c2.category_name,
         c3.category_no, c3.category_name
ORDER BY c1.category_no, c2.category_no, c3.category_no;
```

## 프로시저

```sql
-- 상품입고 프로시저
CREATE OR REPLACE PROCEDURE UP_PRODUCT_IN (
-- 신창만
-- 입고 프로시저. 옵션번호와 수량을 입력하면 해당 상품이 입고된다.
-- 사용법 : EXEC UP_PRODUCT_IN(옵션번호, 수량);
-- ⚠ OPTION_ID 기준이라 PRODUCT_OPTION 행이 없는 상품(option_yn='N')에는 못 씀.
--   그런 상품은 UPDATE PRODUCT SET base_quantity = base_quantity + 수량 WHERE product_no = ... 로 직접 처리
    P_OPTION_NO IN PRODUCT_IN.OPTION_ID%TYPE,
    P_QUANTITY  IN PRODUCT_IN.QUANTITY%TYPE
) IS
BEGIN
    INSERT INTO PRODUCT_IN
    VALUES (SEQ_PRODUCT_IN.NEXTVAL, P_QUANTITY, SYSDATE, P_OPTION_NO);

    COMMIT;
END;
/

-- 회원가입 프로시저
CREATE OR REPLACE PROCEDURE MEMBER_INSERT (
-- 회원 가입
-- EXEC member_insert(회원id, 비밀번호, 이름, 전화번호, 이메일);
-- 신창만
    P_MEMBER_ID     IN MEMBER.MEMBER_ID%TYPE,
    P_MEMBER_PW     IN MEMBER.MEMBER_PW%TYPE,
    P_MEMBER_NAME   IN MEMBER.MEMBER_NAME%TYPE,
    P_PHONE         IN MEMBER.PHONE%TYPE,
    P_EMAIL         IN MEMBER.EMAIL%TYPE
) IS
BEGIN

    VALUES (P_MEMBER_ID, P_MEMBER_PW, P_MEMBER_NAME, P_PHONE, P_EMAIL, '일반회원');

    COMMIT;
END;
/

-- 판매자 가입 프로시저
CREATE OR REPLACE PROCEDURE SELLER_INSERT (
-- 판매자 가입
-- 사용법 : EXEC seller_insert(사업장주소,계좌번호,사업자번호,상호명,은행명, 아이디, 비밀번호, 대표자명, 전화번호, 이메일)
-- 신창만
    P_BUSINESS_ADDRESS      IN SELLER.BUSINESS_ADDRESS%TYPE,
    P_SETTLEMENT_ACCOUNT    IN SELLER.SETTLEMENT_ACCOUNT%TYPE,
    P_BUSINESS_NO           IN SELLER.BUSINESS_NO%TYPE,
    P_STORE_NAME            IN SELLER.STORE_NAME%TYPE,
    P_BANK_NAME             IN SELLER.BANK_NAME%TYPE,
    P_SELLER_ID             IN SELLER.SELLER_ID%TYPE,
    P_SELLER_PW             IN SELLER.SELLER_PW%TYPE,
    P_CEO_NAME              IN SELLER.CEO_NAME%TYPE,
    P_TEL                   IN SELLER.TEL%TYPE,
    P_EMAIL                 IN SELLER.EMAIL%TYPE
) IS
BEGIN
    INSERT INTO SELLER (
        seller_no, business_address, settlement_account, business_no,
        store_name, bank_name, seller_id, seller_pw, ceo_name, tel, email, status
    )
    VALUES (
        SEQ_SELLER.NEXTVAL, P_BUSINESS_ADDRESS, P_SETTLEMENT_ACCOUNT, P_BUSINESS_NO,
        P_STORE_NAME, P_BANK_NAME, P_SELLER_ID, P_SELLER_PW, P_CEO_NAME, P_TEL, P_EMAIL, 0
```

### 1번 상품 조회

```sql
-- PRODUCT_NAME→display_name, PRODUCT_PRICE→base_price, O.PRICE→O.sale_price로 변경.
-- LEFT JOIN으로 바꿔서 옵션이 없는 상품(생수처럼 option_yn='N')도 함께 조회되게 함
-- (그런 상품은 O.* 칼럼이 전부 NULL로 나오고, NVL이 base_price/base_quantity로 대체함)
SELECT
    S.STORE_NAME,
    P.PRODUCT_NO,
    P.display_name,
    O.OPTION_ID,
    O.OPTION1_TYPE,
    O.OPTION1_VALUE,
    O.OPTION2_TYPE,
    O.OPTION2_VALUE,
    O.OPTION3_TYPE,
    O.OPTION3_VALUE,
    NVL(O.sale_price, P.base_price)    AS 판매가,
    NVL(O.QUANTITY, P.base_quantity)   AS 재고,
    O.STATUS                          AS 판매여부
FROM PRODUCT P
JOIN SELLER  S ON P.SELLER_NO  = S.SELLER_NO
LEFT JOIN PRODUCT_OPTION O ON O.PRODUCT_NO = P.PRODUCT_NO
WHERE P.PRODUCT_NO = 1
ORDER BY O.PRODUCT_NO, O.OPTION_ID;
```
