```sql
-- 1번 판매자가 등록한 상품조회
SELECT p.*,
       s.store_name,
       mc.main_category_name,
       md.mid_category_name,
       sc.sub_category_name
FROM PRODUCT p
    JOIN SELLER s
      ON p.seller_no = s.seller_no
    JOIN SUB_CATEGORY sc
      ON p.sub_category_no = sc.sub_category_no
    JOIN MID_CATEGORY md
      ON sc.mid_category_no = md.mid_category_no
    JOIN MAIN_CATEGORY mc
      ON md.main_category_no = mc.main_category_no
WHERE p.seller_no = 1;

-- 1번 판매자의 각 상품별 재고 현황조회
SELECT p.product_no, p.product_name, p.product_desc, p.product_price,
       po.PRICE        AS option_price,
       po.quantity     AS option_quantity,
       po.OPTION_ID,
       po.OPTION1_TYPE,
       po.OPTION1_VALUE,
       po.OPTION2_TYPE,
       po.OPTION2_VALUE,
       po.OPTION3_TYPE,
       po.OPTION3_VALUE,
       po.STATUS       AS option_status
FROM PRODUCT p
JOIN SELLER s
  ON p.seller_no = s.seller_no
LEFT JOIN PRODUCT_OPTION po
  ON p.product_no = po.product_no
WHERE p.seller_no = 8;
```

```sql
DROP PROCEDURE UP_ORDER;

-- 1. ORDERS 테이블용 시퀀스
CREATE SEQUENCE SEQ_ORDER

--- 실행 구문
CREATE SEQUENCE SEQ_PAYMENT
    START WITH 1
    INCREMENT BY 1
    NOCACHE
    NOCYCLE;

CREATE OR REPLACE PROCEDURE UP_ORDER
(   -- 매개변수
    P_MEMBER_NO         IN NUMBER,
    P_ADDRESS_NO        IN NUMBER,
    P_OPTION_ID         IN NUMBER,
    P_ORDER_QTY         IN NUMBER,
    P_PAYMENT_METHOD    IN VARCHAR2
)
IS  -- 변수선언 / type동일한 자료형으로
    V_ORDER_NO          NUMBER;
    V_ORDER_ADDRESS_NO  NUMBER;

    V_PRODUCT_NO        PRODUCT.PRODUCT_NO%TYPE;
    V_SELLER_NO         PRODUCT.SELLER_NO%TYPE;

    V_PRICE             PRODUCT_OPTION.PRICE%TYPE;
    V_STOCK             PRODUCT_OPTION.QUANTITY%TYPE;

    V_TOTAL             NUMBER;
    V_DELIVERY_FEE      NUMBER;

    V_RECEIVER_NAME     DELIVERY_ADDRESS.RECEIVER_NAME%TYPE;
    V_ZIPCODE           DELIVERY_ADDRESS.ZIPCODE%TYPE;
    V_ADDRESS           DELIVERY_ADDRESS.ADDRESS%TYPE;
    V_DETAIL_ADDRESS    DELIVERY_ADDRESS.DETAIL_ADDRESS%TYPE;
    V_REQUEST_MSG       DELIVERY_ADDRESS.REQUEST_MSG%TYPE;

    V_STEP              VARCHAR2(20);

BEGIN
    --------------------------------------------------
    -- 주문수량 체크
    --------------------------------------------------
    IF P_ORDER_QTY <= 0 THEN
        RAISE_APPLICATION_ERROR(-20001,'주문수량은 1개 이상이어야 합니다.');
    END IF;

    --------------------------------------------------
    -- 상품조회 + 판매자 조회
    --------------------------------------------------
    V_STEP := 'PRODUCT';

    SELECT PO.PRODUCT_NO,
           PO.PRICE,
           PO.QUANTITY,
           P.SELLER_NO
    INTO
           V_PRODUCT_NO,
           V_PRICE,
           V_STOCK,
           V_SELLER_NO
    FROM PRODUCT_OPTION PO
         JOIN PRODUCT P
           ON PO.PRODUCT_NO = P.PRODUCT_NO
    WHERE PO.OPTION_ID = P_OPTION_ID
    FOR UPDATE;

    IF V_STOCK < P_ORDER_QTY THEN
        RAISE_APPLICATION_ERROR(-20002,'재고가 부족합니다.');
    END IF;

    --------------------------------------------------
    -- 배송지 조회
    --------------------------------------------------
    V_STEP := 'ADDRESS';

    SELECT RECEIVER_NAME,
           ZIPCODE,
           ADDRESS,
           DETAIL_ADDRESS,
           REQUEST_MSG
    INTO
           V_RECEIVER_NAME,
           V_ZIPCODE,
           V_ADDRESS,
           V_DETAIL_ADDRESS,
           V_REQUEST_MSG
    FROM DELIVERY_ADDRESS
    WHERE ADDRESS_NO = P_ADDRESS_NO
      AND MEMBER_NO = P_MEMBER_NO;

    --------------------------------------------------
    -- 주문번호 생성
    --------------------------------------------------
    SELECT SEQ_ORDER.NEXTVAL
    INTO V_ORDER_NO
    FROM DUAL;

    --------------------------------------------------
    -- 배송비 계산
    --------------------------------------------------
    V_TOTAL := V_PRICE * P_ORDER_QTY;

    IF V_TOTAL >= 20000 THEN
        V_DELIVERY_FEE := 0;
    ELSE
        V_DELIVERY_FEE := 3000;
    END IF;

    V_TOTAL := V_TOTAL + V_DELIVERY_FEE;

    --------------------------------------------------
    -- 주문주소 저장
    --------------------------------------------------
    INSERT INTO ORDER_ADDRESS
    (
        RECEIVER_NAME,
        ZIPCODE,
        ADDRESS,
        DETAIL_ADDRESS,
        REQUEST_MSG
    )
    VALUES
    (
        V_RECEIVER_NAME,
        V_ZIPCODE,
        V_ADDRESS,
        V_DETAIL_ADDRESS,
        V_REQUEST_MSG
    )
    RETURNING ORDER_ADDRESS_NO
    INTO V_ORDER_ADDRESS_NO;

    --------------------------------------------------
    -- 주문 생성
    --------------------------------------------------
    INSERT INTO ORDERS
    (
        ORDER_NO,
        MEMBER_NO,
        ORDER_ADDRESS_NO,
        ORDER_DATE,
        PAYMENT_METHOD,
        DELIVERY_FEE,
        TOTAL_PRICE,
        ORDER_STATUS
    )
    VALUES
    (
        V_ORDER_NO,
        P_MEMBER_NO,
        V_ORDER_ADDRESS_NO,
        SYSDATE,
        P_PAYMENT_METHOD,
        V_DELIVERY_FEE,
        V_TOTAL,
        '주문접수'
    );

    --------------------------------------------------
    -- 주문상세 생성
    --------------------------------------------------
    INSERT INTO ORDER_DETAIL
    (
        ORDER_DETAIL_NO,
        ORDER_NO,
        PRODUCT_NO,
        ORDER_QTY,
        PRICE,
        OPTION_ID
    )
    VALUES
    (
        SEQ_ORDER_DETAIL.NEXTVAL,
        V_ORDER_NO,
        V_PRODUCT_NO,
        P_ORDER_QTY,
        V_PRICE,
        P_OPTION_ID
    );

    --------------------------------------------------
    -- 재고 감소
    --------------------------------------------------
    UPDATE PRODUCT_OPTION
    SET QUANTITY = QUANTITY - P_ORDER_QTY
    WHERE OPTION_ID = P_OPTION_ID;

    --------------------------------------------------
    -- 재고이력 저장
    --------------------------------------------------
    INSERT INTO STOCK_HISTORY
    (
        HISTORY_ID,
        SELLER_NO,
        OPTION_ID,
        CHANGE_TYPE,
        QUANTITY,
        REASON,
        REG_DATE
    )
    VALUES
    (
        SEQ_STOCK_HISTORY.NEXTVAL,
        V_SELLER_NO,
        P_OPTION_ID,
        '판매',
        P_ORDER_QTY,
        '주문번호 : ' || V_ORDER_NO,
        SYSDATE
    );

    --------------------------------------------------
    -- 결제 생성
    --------------------------------------------------
    INSERT INTO PAYMENT
    (
        PAYMENT_NO,
        PAYMENT_DATE,
        PAYMENT_AMOUNT,
        ORDER_NO
    )
    VALUES
    (
        SEQ_PAYMENT.NEXTVAL,
        SYSDATE,
        V_TOTAL,
        V_ORDER_NO
    );

    --------------------------------------------------
    -- 배송 생성
    --------------------------------------------------
    INSERT INTO DELIVERY
    (
        DELIVERY_NO,
        ORDER_NO,
        DELIVERY_SERVICE_CODE,
        DELIVERY_STATUS,
        DELIVERY_START_DATE,
        DELIVERY_END_DATE
    )
    VALUES
    (
        SEQ_DELIVERY.NEXTVAL,
        V_ORDER_NO,
        'CJ대한통운',
        '주문접수',
        SYSDATE,
        SYSDATE + 2
    );

    COMMIT;

EXCEPTION
    WHEN NO_DATA_FOUND THEN
        ROLLBACK;

        IF V_STEP = 'PRODUCT' THEN
            RAISE_APPLICATION_ERROR(-20003,'상품 옵션이 존재하지 않습니다.');
        ELSIF V_STEP = 'ADDRESS' THEN
            RAISE_APPLICATION_ERROR(-20004,'배송지가 존재하지 않습니다.');
        ELSE
            RAISE_APPLICATION_ERROR(-20005,'데이터를 찾을 수 없습니다.');
        END IF;

    WHEN OTHERS THEN
        ROLLBACK;
        RAISE;

END;
/
```

```sql
-- 위시리스트 프로시저
CREATE OR REPLACE PROCEDURE IN_WISHLIST
(
    p_MEMBER_NO IN WISHLIST.MEMBER_NO%TYPE,
    p_OPTION_ID IN WISHLIST.OPTION_ID%TYPE
)
IS
    V_CHECK_NO NUMBER;
BEGIN
--중복입력 방지(WISHLIST에 PK없음)
    SELECT COUNT(*)
    INTO V_CHECK_NO
    FROM WISHLIST
    WHERE MEMBER_NO = p_MEMBER_NO
      AND OPTION_ID = p_OPTION_ID;

    IF V_CHECK_NO = 0 THEN
        INSERT INTO WISHLIST
        (
            MEMBER_NO,
            OPTION_ID
        )
        VALUES
        (
            p_MEMBER_NO,
            p_OPTION_ID
        );

        DBMS_OUTPUT.PUT_LINE('찜 목록에 추가되었습니다.');
    ELSE
        --찜 취소
        DELETE FROM WISHLIST
        WHERE MEMBER_NO = P_MEMBER_NO
          AND OPTION_ID = P_OPTION_ID;

        DBMS_OUTPUT.PUT_LINE('찜 목록에서 삭제되었습니다.');
    END IF;

EXCEPTION
    WHEN OTHERS THEN
        DBMS_OUTPUT.PUT_LINE(SQLERRM);
END;
/

--장바구니 프로시저 생성
CREATE OR REPLACE PROCEDURE IN_CART
(
    p_MEMBER_NO IN CART.MEMBER_NO%TYPE,
    p_OPTION_ID IN CART.OPTION_ID%TYPE,
    p_QUANTITY  IN CART.QUANTITY%TYPE
)
IS
    V_COUNT NUMBER;
BEGIN
    --중복입력 방지(CART에 PK없음)
    SELECT COUNT(*)
    INTO V_COUNT
    FROM CART
    WHERE MEMBER_NO = p_MEMBER_NO
      AND OPTION_ID = p_OPTION_ID;

    IF V_COUNT = 0 THEN
        INSERT INTO CART
        (
            MEMBER_NO,
            OPTION_ID,
            QUANTITY
        )
        VALUES
        (
            p_MEMBER_NO,
            p_OPTION_ID,
            p_QUANTITY
        );

    ELSE
        -- 현재 수량 기준 조회
        DECLARE
            V_QTY CART.QUANTITY%TYPE;
        BEGIN
            SELECT QUANTITY
            INTO V_QTY
            FROM CART
            WHERE MEMBER_NO = p_MEMBER_NO
              AND OPTION_ID = p_OPTION_ID;

            V_QTY := V_QTY + p_QUANTITY;

            -- 2) 0 이하 → DELETE
            IF V_QTY <= 0 THEN
                DELETE FROM CART
                WHERE MEMBER_NO = p_MEMBER_NO
                  AND OPTION_ID = p_OPTION_ID;

            -- 3) 증가/감소 → UPDATE
            ELSE
                UPDATE CART
                SET QUANTITY = V_QTY
                WHERE MEMBER_NO = p_MEMBER_NO
                  AND OPTION_ID = p_OPTION_ID;
            END IF;
        END;
    END IF;

EXCEPTION
    WHEN OTHERS THEN
        DBMS_OUTPUT.PUT_LINE(SQLERRM);
END;
/

--실행(회원 아이디, 옵션 아이디, 수량)

--확인
```

```sql
--VIEW를 사용한 유저별 결제내역 조회
CREATE OR REPLACE VIEW V_PAYMENT_HISTORY AS
SELECT
    M.MEMBER_NO        AS 회원번호,
    M.MEMBER_NAME       AS 회원명,
    O.ORDER_NO          AS 주문번호,
    P.PAYMENT_DATE      AS 결제일,
    PR.PRODUCT_NAME     AS 상품명,
    PO.OPTION1_VALUE    AS 옵션,
    OD.ORDER_QTY        AS 수량,
    OD.PRICE            AS 구매금액,
    P.PAYMENT_AMOUNT    AS 결제금액,
    O.ORDER_STATUS      AS 주문상태
FROM MEMBER M
JOIN ORDERS O
    ON M.MEMBER_NO = O.MEMBER_NO
JOIN PAYMENT P
    ON O.ORDER_NO = P.ORDER_NO
JOIN ORDER_DETAIL OD
    ON O.ORDER_NO = OD.ORDER_NO
JOIN PRODUCT PR
    ON OD.PRODUCT_NO = PR.PRODUCT_NO
JOIN PRODUCT_OPTION PO
    ON OD.OPTION_ID = PO.OPTION_ID;

--조회
SELECT *
FROM V_PAYMENT_HISTORY
ORDER BY 주문번호 DESC;
```

```sql
-- 리뷰 작성 프로시저 --
-----------------------------------------------------------------------------
DROP PROCEDURE add_review;
-----------------------------------------------------------------------------

CREATE OR REPLACE PROCEDURE add_review (
    p_order_detail_no   IN NUMBER,
    p_rating            IN NUMBER,
    p_content           IN VARCHAR2
)
IS
--커서(Cursor) 주문상세 + 주문 조인
    CURSOR c_order IS
        SELECT od.order_no, od.product_no, od.option_id,
               o.member_no, o.order_status
        FROM order_detail od
        JOIN orders o ON od.order_no = o.order_no
        WHERE od.order_detail_no = p_order_detail_no;

    v_row c_order%ROWTYPE;
    v_cnt NUMBER;

BEGIN
-- 1. 주문상세 + 주문 JOIN 조회 --
    OPEN c_order;
    FETCH c_order INTO v_row;
    IF c_order%NOTFOUND THEN
        CLOSE c_order;
        RAISE_APPLICATION_ERROR(-20001, '해당 주문이 존재하지 않습니다: ' || p_order_detail_no);
    END IF;
    CLOSE c_order;

-- 2. 중복 방지 : 같은 주문 + 같은 상품 + 같은 옵션 + 리뷰 중복 안되게 예외 처리 --
    SELECT COUNT(*)
      INTO v_cnt
      FROM review r
      JOIN order_detail od ON r.order_detail_no = od.order_detail_no
     WHERE od.order_no    = v_row.order_no
       AND od.product_no  = v_row.product_no
       AND (od.option_id = v_row.option_id
            OR (od.option_id IS NULL AND v_row.option_id IS NULL));

    IF v_cnt > 0 THEN
        RAISE_APPLICATION_ERROR(-20002, '이미 리뷰가 작성된 상품입니다.');
    END IF;

-- 3. 평점 1~5점 제한 --
    IF p_rating NOT BETWEEN 1 AND 5 THEN
        RAISE_APPLICATION_ERROR(-20003, '평점은 1~5 사이만 가능합니다.');
    END IF;

-- 4. 배송완료 후 리뷰 작성 가능 --
    IF v_row.order_status != '배송완료' THEN
        RAISE_APPLICATION_ERROR(-20004, '배송완료된 주문만 리뷰 작성 가능합니다. (현재: ' || v_row.order_status || ')');
    END IF;

-- 5. 리뷰 INSERT 확인 출력까지 --
    INSERT INTO review (review_no, rating, review_content, review_date, member_no, order_detail_no)
    VALUES (seq_review_no.NEXTVAL, p_rating, p_content, SYSDATE, v_row.member_no, p_order_detail_no);
    COMMIT;
    DBMS_OUTPUT.PUT_LINE('리뷰가 등록 되었습니다. (회원 ' || v_row.member_no || ', 주문상세 ' || p_order_detail_no || ')');

--EXCEPTION
END;
/

COMMIT;
-----------------------------------------------------------------------------
```
