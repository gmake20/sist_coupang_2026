# 주문 - 양인석

## 주문

```sql
DROP PROCEDURE UP_ORDER;

-- 1. ORDERS 테이블용 시퀀스
CREATE SEQUENCE SEQ_ORDER
    START WITH 1
    INCREMENT BY 1
    NOMAXVALUE
    NOMINVALUE
    NOCYCLE
    NOCACHE;

-- 2. ORDER_DETAIL 테이블용 시퀀스
CREATE SEQUENCE SEQ_ORDER_DETAIL
    START WITH 1
    INCREMENT BY 1
    NOMAXVALUE
    NOMINVALUE
    NOCYCLE
    NOCACHE;

-- 3. PAYMENT테이블용 시퀀스
CREATE SEQUENCE SEQ_PAYMENT
    START WITH 1
    INCREMENT BY 1
    NOCACHE
    NOCYCLE;

-- 4. DELIVERY 테이블용 시퀀스
CREATE SEQUENCE SEQ_DELIVERY
    START WITH 1
    INCREMENT BY 1
    NOMAXVALUE
    NOMINVALUE
    NOCYCLE
    NOCACHE;


--- 실행 구문
CREATE SEQUENCE SEQ_PAYMENT
    START WITH 1
    INCREMENT BY 1
    NOCACHE
    NOCYCLE;

CREATE OR REPLACE PROCEDURE UP_ORDER
(   -- 매개변수
    P_MEMBER_NO         IN NUMBER,
    P_ADDRESS_NO        IN NUMBER,
    P_OPTION_ID         IN NUMBER,
    P_ORDER_QTY         IN NUMBER,
    P_PAYMENT_METHOD    IN VARCHAR2
)
IS  -- 변수선언 / type동일한 자료형으로
    V_ORDER_NO          NUMBER;
    V_ORDER_ADDRESS_NO  NUMBER;

    V_PRODUCT_NO        PRODUCT.PRODUCT_NO%TYPE;
    V_SELLER_NO         PRODUCT.SELLER_NO%TYPE;

    V_PRICE             PRODUCT_OPTION.PRICE%TYPE;
    V_STOCK             PRODUCT_OPTION.QUANTITY%TYPE;

    V_TOTAL             NUMBER;
    V_DELIVERY_FEE      NUMBER;

    V_RECEIVER_NAME     DELIVERY_ADDRESS.RECEIVER_NAME%TYPE;
    V_ZIPCODE           DELIVERY_ADDRESS.ZIPCODE%TYPE;
    V_ADDRESS           DELIVERY_ADDRESS.ADDRESS%TYPE;
    V_DETAIL_ADDRESS    DELIVERY_ADDRESS.DETAIL_ADDRESS%TYPE;
    V_REQUEST_MSG       DELIVERY_ADDRESS.REQUEST_MSG%TYPE;

    V_STEP              VARCHAR2(20);

BEGIN
    --------------------------------------------------
    -- 주문수량 체크
    --------------------------------------------------
    IF P_ORDER_QTY <= 0 THEN
        RAISE_APPLICATION_ERROR(-20001,'주문수량은 1개 이상이어야 합니다.');
    END IF;

    --------------------------------------------------
    -- 상품조회 + 판매자 조회
    --------------------------------------------------
    V_STEP := 'PRODUCT';

    SELECT PO.PRODUCT_NO,
           PO.PRICE,
           PO.QUANTITY,
           P.SELLER_NO
    INTO
           V_PRODUCT_NO,
           V_PRICE,
           V_STOCK,
           V_SELLER_NO
    FROM PRODUCT_OPTION PO
         JOIN PRODUCT P
           ON PO.PRODUCT_NO = P.PRODUCT_NO
    WHERE PO.OPTION_ID = P_OPTION_ID
    FOR UPDATE;

    IF V_STOCK < P_ORDER_QTY THEN
        RAISE_APPLICATION_ERROR(-20002,'재고가 부족합니다.');
    END IF;

    --------------------------------------------------
    -- 배송지 조회
    --------------------------------------------------
    V_STEP := 'ADDRESS';

    SELECT RECEIVER_NAME,
           ZIPCODE,
           ADDRESS,
           DETAIL_ADDRESS,
           REQUEST_MSG
    INTO
           V_RECEIVER_NAME,
           V_ZIPCODE,
           V_ADDRESS,
           V_DETAIL_ADDRESS,
           V_REQUEST_MSG
    FROM DELIVERY_ADDRESS
    WHERE ADDRESS_NO = P_ADDRESS_NO
      AND MEMBER_NO = P_MEMBER_NO;

    --------------------------------------------------
    -- 주문번호 생성
    --------------------------------------------------
    SELECT SEQ_ORDER.NEXTVAL
    INTO V_ORDER_NO
    FROM DUAL;

    --------------------------------------------------
    -- 배송비 계산
    --------------------------------------------------
    V_TOTAL := V_PRICE * P_ORDER_QTY;

    IF V_TOTAL >= 20000 THEN
        V_DELIVERY_FEE := 0;
    ELSE
        V_DELIVERY_FEE := 3000;
    END IF;

    V_TOTAL := V_TOTAL + V_DELIVERY_FEE;

    --------------------------------------------------
    -- 주문주소 저장
    --------------------------------------------------
    INSERT INTO ORDER_ADDRESS
    (
        RECEIVER_NAME,
        ZIPCODE,
        ADDRESS,
        DETAIL_ADDRESS,
        REQUEST_MSG
    )
    VALUES
    (
        V_RECEIVER_NAME,
        V_ZIPCODE,
        V_ADDRESS,
        V_DETAIL_ADDRESS,
        V_REQUEST_MSG
    )
    RETURNING ORDER_ADDRESS_NO
    INTO V_ORDER_ADDRESS_NO;

    --------------------------------------------------
    -- 주문 생성
    --------------------------------------------------
    INSERT INTO ORDERS
    (
        ORDER_NO,
        MEMBER_NO,
        ORDER_ADDRESS_NO,
        ORDER_DATE,
        PAYMENT_METHOD,
        DELIVERY_FEE,
        TOTAL_PRICE,
        ORDER_STATUS
    )
    VALUES
    (
        V_ORDER_NO,
        P_MEMBER_NO,
        V_ORDER_ADDRESS_NO,
        SYSDATE,
        P_PAYMENT_METHOD,
        V_DELIVERY_FEE,
        V_TOTAL,
        '주문접수'
    );

    --------------------------------------------------
    -- 주문상세 생성
    --------------------------------------------------
    INSERT INTO ORDER_DETAIL
    (
        ORDER_DETAIL_NO,
        ORDER_NO,
        PRODUCT_NO,
        ORDER_QTY,
        PRICE,
        OPTION_ID
    )
    VALUES
    (
        SEQ_ORDER_DETAIL.NEXTVAL,
        V_ORDER_NO,
        V_PRODUCT_NO,
        P_ORDER_QTY,
        V_PRICE,
        P_OPTION_ID
    );

    --------------------------------------------------
    -- 재고 감소
    --------------------------------------------------
    UPDATE PRODUCT_OPTION
    SET QUANTITY = QUANTITY - P_ORDER_QTY
    WHERE OPTION_ID = P_OPTION_ID;

    --------------------------------------------------
    -- 재고이력 저장
    --------------------------------------------------
    INSERT INTO STOCK_HISTORY
    (
        HISTORY_ID,
        SELLER_NO,
        OPTION_ID,
        CHANGE_TYPE,
        QUANTITY,
        REASON,
        REG_DATE
    )
    VALUES
    (
        SEQ_STOCK_HISTORY.NEXTVAL,
        V_SELLER_NO,
        P_OPTION_ID,
        '판매',
        P_ORDER_QTY,
        '주문번호 : ' || V_ORDER_NO,
        SYSDATE
    );

    --------------------------------------------------
    -- 결제 생성
    --------------------------------------------------
    INSERT INTO PAYMENT
    (
        PAYMENT_NO,
        PAYMENT_DATE,
        PAYMENT_AMOUNT,
        ORDER_NO
    )
    VALUES
    (
        SEQ_PAYMENT.NEXTVAL,
        SYSDATE,
        V_TOTAL,
        V_ORDER_NO
    );

    --------------------------------------------------
    -- 배송 생성
    --------------------------------------------------
    INSERT INTO DELIVERY
    (
        DELIVERY_NO,
        ORDER_NO,
        DELIVERY_SERVICE_CODE,
        DELIVERY_STATUS,
        DELIVERY_START_DATE,
        DELIVERY_END_DATE
    )
    VALUES
    (
        SEQ_DELIVERY.NEXTVAL,
        V_ORDER_NO,
        'CJ대한통운',
        '주문접수',
        SYSDATE,
        SYSDATE + 2
    );

    COMMIT;

EXCEPTION

    WHEN NO_DATA_FOUND THEN
        ROLLBACK;

        IF V_STEP = 'PRODUCT' THEN
            RAISE_APPLICATION_ERROR(-20003,'상품 옵션이 존재하지 않습니다.');
        ELSIF V_STEP = 'ADDRESS' THEN
            RAISE_APPLICATION_ERROR(-20004,'배송지가 존재하지 않습니다.');
        ELSE
            RAISE_APPLICATION_ERROR(-20005,'데이터를 찾을 수 없습니다.');
        END IF;

    WHEN OTHERS THEN
        ROLLBACK;
        RAISE;

END;
/

-----------------------------------------------------------------------------
--- 실행테스트구문
EXEC UP_ORDER(2, 2, 14, 5, '신용카드');
-- 이상섭,서울특별시 강남구 논현동 106-7,
--(회원번호, 배송지번호, 옵션ID, 주문수량, 결제방법)
```

## 프로시저 설명

주문이 들어왔을 때 상품 조회 → 재고 확인 → 배송지 확인 → 주문 생성 → 주문 상세 생성 → 재고 차감 → 배송 생성까지 하나의 트랜잭션으로 처리하는 주문 처리 로직

**주문 요청 → 주문수량 검사 → 상품 조회 및 재고 확인 → 배송지 조회 → 주문번호 생성 → 배송비 및 총금액 계산 → 주문배송지 저장 → 주문 생성 → 주문상세 생성 → 재고 감소 → 재고이력(STOCK_HISTORY) 저장 → 결제 생성 → 배송 생성 → COMMIT**
(주문, 결제, 배송, 재고 처리)

▼ 회원번호, 배송지, 상품옵션, 주문수량, 결제수단을 입력받기

```sql
P_MEMBER_NO
P_ADDRESS_NO
P_OPTION_ID
P_ORDER_QTY
P_PAYMENT_METHOD
```

▼ 주문 수량이 1개 미만이면 비정상 요청이기 때문에 즉시 에러 처리함

```sql
IF P_ORDER_QTY <= 0 THEN
```

▼ 상품 정보를 조회하면서 FOR UPDATE를 사용해 해당 상품을 잠금 처리하여 동시에 여러 주문이 들어왔을 때 재고 꼬임을 방지!

```sql
FROM PRODUCT_OPTION
FOR UPDATE
```

▼ 현재 재고보다 주문 수량이 많으면 주문을 중단!

```sql
IF V_STOCK < P_ORDER_QTY
```

▼ 해당 회원의 배송지 정보를 가져와서 주문 주소로 사용!

```sql
FROM DELIVERY_ADDRESS
```

▼ 주문 고유번호를 시퀀스로 생성하고 단가 × 수량으로 총 결제 금액을 계산

```sql
V_ORDER_NO := SEQ_ORDER.NEXTVAL;
V_TOTAL := V_PRICE * P_ORDER_QTY;
```

배송지 정보를 ORDER_ADDRESS 테이블에 저장하고 생성된 주소 번호를 주문에 연결함

▼ 주문 기본 정보를 저장하며 주문 상태는 "주문접수"로 시작

```sql
INSERT INTO ORDERS
```

▼ 실제 상품 정보(수량, 가격, 옵션)를 저장

```sql
INSERT INTO ORDER_DETAIL
```

▼ 주문이 완료되면 해당 상품의 재고를 차감

```sql
UPDATE PRODUCT_OPTION
```

▼ 주문과 동시에 배송 데이터도 생성하며 시작일과 종료일은 SYSDATE로 기록

```sql
INSERT INTO DELIVERY
```

▼ 모든 작업이 성공해야만 한 번에 저장되는 구조

```sql
COMMIT;
```

▼ 상품이나 배송지가 없거나 문제가 생기면 ROLLBACK으로 모든 작업을 취소하고 에러를 반환함

```sql
EXCEPTION
```

## 주문 취소 (ORDER_STATUS = '취소')

```sql
UPDATE ORDERS
SET ORDER_STATUS = '취소'
WHERE ORDER_NO = :ORDER_NO
  AND ORDER_STATUS = '주문접수';

-- 재고 복구
UPDATE PRODUCT_OPTION
SET QUANTITY = QUANTITY + :ORDER_QTY
WHERE OPTION_ID = :OPTION_ID;
```

## 환불 (결제 취소)

```sql
UPDATE PAYMENT
SET PAYMENT_STATUS = '환불'
WHERE ORDER_NO = :ORDER_NO;

UPDATE ORDERS
SET ORDER_STATUS = '취소'
WHERE ORDER_NO = :ORDER_NO;
```

## 반품 (배송 완료 후)

```sql
UPDATE ORDERS
SET ORDER_STATUS = '반품'
WHERE ORDER_NO = :ORDER_NO
  AND ORDER_STATUS = '배송완료';

-- 반품 입고
UPDATE PRODUCT_OPTION
SET QUANTITY = QUANTITY + :ORDER_QTY
WHERE OPTION_ID = :OPTION_ID;
```

## 교환

```sql
UPDATE ORDERS
SET ORDER_STATUS = '교환요청'
WHERE ORDER_NO = :ORDER_NO;

-- 재고 복구
UPDATE PRODUCT_OPTION
SET QUANTITY = QUANTITY + :ORDER_QTY
WHERE OPTION_ID = :OPTION_ID;
```
