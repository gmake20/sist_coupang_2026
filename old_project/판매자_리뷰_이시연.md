## 리뷰

```sql
-----------------------------------------------------------------------------
--리뷰--

-- 1. 특정 상품의 리뷰 목록 (별점, 작성자, 내용, 날짜)
SELECT r.review_no,
       p.product_name,
       m.member_name,
       r.rating,
       r.review_content,
       TO_CHAR(r.review_date, 'YYYY-MM-DD') AS "작성일"
FROM review r
    JOIN order_detail od ON r.order_detail_no = od.order_detail_no
    JOIN product p        ON od.product_no = p.product_no
    JOIN member m         ON r.member_no = m.member_no
WHERE p.product_no = 1
ORDER BY r.review_date DESC;

-- 2. 상품별 평균 별점, 리뷰 수
SELECT p.product_no,
       p.product_name,
       COUNT(*)                AS "리뷰 수",
       ROUND(AVG(r.rating),1)  AS "평균 별점"
FROM review r
    JOIN order_detail od ON r.order_detail_no = od.order_detail_no
    JOIN product p        ON od.product_no = p.product_no
GROUP BY p.product_no, p.product_name
ORDER BY "평균 별점" DESC, "리뷰 수" DESC;

-- 3. 특정 회원이 쓴 리뷰 목록
SELECT m.member_no, m.member_id, m.member_name, m.rank
     , r.review_no, r.rating, r.review_content, r.review_date
FROM review r
    JOIN member m ON r.member_no = m.member_no
WHERE r.member_no = 1
ORDER BY r.review_date DESC;

-----------------------------------------------------------------------------
SELECT * FROM review;
-----------------------------------------------------------------------------
```

## 리뷰 작성 프로시저

```sql
-----------------------------------------------------------------------------
DROP PROCEDURE add_review;
-----------------------------------------------------------------------------

CREATE OR REPLACE PROCEDURE add_review (
    p_order_detail_no   IN NUMBER,
    p_rating            IN NUMBER,
    p_content           IN VARCHAR2
)
IS
--커서(Cursor) 주문상세 + 주문 조인
    CURSOR c_order IS
        SELECT od.order_no, od.product_no, od.option_id,
               o.member_no, o.order_status
        FROM order_detail od
        JOIN orders o ON od.order_no = o.order_no
        WHERE od.order_detail_no = p_order_detail_no;

    v_row c_order%ROWTYPE;
    v_cnt NUMBER;

BEGIN
-- 1. 주문상세 + 주문 JOIN 조회 --
    OPEN c_order;
    FETCH c_order INTO v_row;
    IF c_order%NOTFOUND THEN
        CLOSE c_order;
        RAISE_APPLICATION_ERROR(-20001, '해당 주문이 존재하지 않습니다: ' || p_order_detail_no);
    END IF;
    CLOSE c_order;

-- 2. 중복 방지 : 같은 주문 + 같은 상품 + 같은 옵션 + 리뷰 중복 안되게 예외 처리 --
    SELECT COUNT(*)
      INTO v_cnt
      FROM review r
      JOIN order_detail od ON r.order_detail_no = od.order_detail_no
     WHERE od.order_no    = v_row.order_no
       AND od.product_no  = v_row.product_no
       AND (od.option_id = v_row.option_id
            OR (od.option_id IS NULL AND v_row.option_id IS NULL));

    IF v_cnt > 0 THEN
        RAISE_APPLICATION_ERROR(-20002, '이미 리뷰가 작성된 상품입니다.');
    END IF;

-- 3. 평점 1~5점 제한 --
    IF p_rating NOT BETWEEN 1 AND 5 THEN
        RAISE_APPLICATION_ERROR(-20003, '평점은 1~5 사이만 가능합니다.');
    END IF;

-- 4. 배송완료 후 리뷰 작성 가능 --
    IF v_row.order_status != '배송완료' THEN
        RAISE_APPLICATION_ERROR(-20004, '배송완료된 주문만 리뷰 작성 가능합니다. (현재: ' || v_row.order_status || ')');
    END IF;

-- 5. 리뷰 INSERT 확인 출력까지 --
    INSERT INTO review (review_no, rating, review_content, review_date, member_no, order_detail_no)
    VALUES (seq_review_no.NEXTVAL, p_rating, p_content, SYSDATE, v_row.member_no, p_order_detail_no);
    COMMIT;
    DBMS_OUTPUT.PUT_LINE('리뷰가 등록 되었습니다. (회원 ' || v_row.member_no || ', 주문상세 ' || p_order_detail_no || ')');

--EXCEPTION
END;
/

COMMIT;
```

## 테스트 실행

```sql
-----------------------------------------------------------------------------
-- 1. 주문 생성 : 프로시저(팀원 양인석) --
EXEC up_order(1, 1, 1, 2, '카드');
EXEC up_order(2, 2, 3, 1, '카드');
EXEC up_order(3, 3, 5, 3, '계좌이체');
EXEC up_order(1, 1, 2, 1, '카드');
EXEC up_order(2, 2, 7, 2, '쿠페이');
EXEC up_order(3, 3, 4, 1, '카드');

-- 2. 배송 완료 UPDATE문으로 처리 --
UPDATE delivery
SET delivery_status = '배송완료'
WHERE order_no IN ( 1,2,3,4,5,6 );

UPDATE orders
SET order_status = '배송완료'
WHERE order_no IN ( 1,2,3,4,5,6 );

COMMIT;

-- 3. 리뷰 작성 프로시저 --
-- 정상 리뷰 등록 --
EXEC add_review(1, 5, '상품 너무 좋아요!');
EXEC add_review(2, 3, '기대에 못 미쳤습니다.');
EXEC add_review(3, 5, '배송이 정말 빨라요.');
EXEC add_review(4, 2, '잘받았습니다.');
EXEC add_review(5, 2, '좋아요');

-- 에러 --
-- -20001 해당 주문이 존재하지 않습니다.
EXEC add_review(999, 5, '테스트');
-- -20002 중복
EXEC add_review(2, 5, '중복 테스트');
-- -20003 평점 범위
EXEC add_review(6, 9, '평점 테스트');
-- -20004 배송 전 리뷰 작성
EXEC add_review(7, 5, '배송 전 리뷰 테스트');

SELECT * FROM order_detail;
SELECT * FROM delivery;
SELECT * FROM review;
```
